{{- /* Shortcode: deeplink-gitlab-issue Renders a link to GitLab's "new work item" page with prefilled values. Parameters (all named): baseURL — REQUIRED. Project or group URL, e.g. https://gitlab.com/g/p title — OPTIONAL. Issue title description — OPTIONAL. Issue body (Markdown) issueType — OPTIONAL. "issue" or "incident" confidential — OPTIONAL. "true" to mark the issue as confidential template — OPTIONAL. Project description template name descriptionTemplate — OPTIONAL. description_template parameter (GitLab 17.9+) addRelatedIssue — OPTIONAL. Numeric ID of an existing issue to relate to text — OPTIONAL. Link text (defaults to the title, then to "File an issue") class — OPTIONAL. CSS class to add to the tag target — OPTIONAL. Target attribute (e.g. "_blank") rel — OPTIONAL. Rel attribute (e.g. "noopener") Example: {{< deeplink-gitlab-issue baseURL="https://gitlab.com/g/p" title="Bug: login broken" description="Steps to reproduce..." >}} NOTE: This template mirrors gitlab.Issue() in the Go library and produces the same URL. Two details matter: - The `issue[...]` keys are written literally, because GitLab expects `issue[title]`, not `issue%5Btitle%5D`. - Values are encoded with `replace (urlquery .) "+" "%20"`, which is Hugo's `urlquery` (identical to Go's url.QueryEscape) with spaces written as %20 rather than +, matching the format GitLab documents. If you change one, change the other. */ -}} {{- $baseURL := .Get "baseURL" -}} {{- if not $baseURL -}} {{- errorf "deeplink-gitlab-issue: baseURL is required" -}} {{- end -}} {{- $pairs := slice -}} {{- with .Get "title" -}} {{- $pairs = $pairs | append (printf "issue[title]=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- with .Get "description" -}} {{- $pairs = $pairs | append (printf "issue[description]=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- with .Get "issueType" -}} {{- $pairs = $pairs | append (printf "issue[issue_type]=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- if eq (.Get "confidential") "true" -}} {{- $pairs = $pairs | append "issue[confidential]=true" -}} {{- end -}} {{- with .Get "template" -}} {{- $pairs = $pairs | append (printf "issuable_template=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- with .Get "descriptionTemplate" -}} {{- $pairs = $pairs | append (printf "description_template=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- with .Get "addRelatedIssue" -}} {{- if gt (int .) 0 -}} {{- $pairs = $pairs | append (printf "add_related_issue=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- end -}} {{- $query := delimit $pairs "&" -}} {{- $url := printf "%s/-/work_items/new" (strings.TrimRight "/" $baseURL) -}} {{- if $query -}}{{- $url = printf "%s?%s" $url $query -}}{{- end -}} {{- $text := or (.Get "text") (.Get "title") "File an issue" -}} {{ $text }}
