{{- /* Shortcode: deeplink-teams-channel Renders a link that opens an existing Microsoft Teams channel. Parameters (all named): channelId — REQUIRED. Channel identifier, e.g. "19:@thread.tacv2" channelName — OPTIONAL. Human-readable channel name (used in URL path) groupId — OPTIONAL. Microsoft 365 group ID tenantId — OPTIONAL. Microsoft Entra tenant ID private — OPTIONAL. "true" for a private (next-gen) channel shared — OPTIONAL. "true" for a shared (cross-tenant) channel text — OPTIONAL. Link text class — OPTIONAL. CSS class target — OPTIONAL. Target attribute rel — OPTIONAL. Rel attribute Example: {{< deeplink-teams-channel channelId="19:abc@thread.tacv2" channelName="General" groupId="..." tenantId="..." >}} NOTE: This template mirrors teams.Channel() in the Go library and produces the same URL. The path segments use `replace (urlquery .) "+" "%20"` — NOT Hugo's `urlize`, which would lowercase the channel name and turn spaces into hyphens. If you change one, change the other. */ -}} {{- $channelID := .Get "channelId" -}} {{- if not $channelID -}} {{- errorf "deeplink-teams-channel: channelId is required" -}} {{- end -}} {{- $path := printf "https://teams.microsoft.com/l/channel/%s" (replace (urlquery $channelID) "+" "%20") -}} {{- with .Get "channelName" -}} {{- $path = printf "%s/%s" $path (replace (urlquery .) "+" "%20") -}} {{- end -}} {{- $pairs := slice -}} {{- with .Get "groupId" -}} {{- $pairs = $pairs | append (printf "groupId=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- with .Get "tenantId" -}} {{- $pairs = $pairs | append (printf "tenantId=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- if eq (.Get "private") "true" -}} {{- $pairs = $pairs | append "ngc=true" -}} {{- end -}} {{- if eq (.Get "shared") "true" -}} {{- $pairs = $pairs | append "allowXTenantAccess=true" -}} {{- end -}} {{- $query := delimit $pairs "&" -}} {{- $url := $path -}} {{- if $query -}}{{- $url = printf "%s?%s" $url $query -}}{{- end -}} {{- $text := or (.Get "text") (.Get "channelName") "Open channel" -}} {{ $text }}
