CSS class target — OPTIONAL. Target attribute rel — OPTIONAL. "+" "%20")) -}} {{- end -}} {{- $query := delimit $pairs "&" -}} {{- $url := "https://outlook.office.com/calendar/deeplink/compose" -}} {{- if $query -}}{{- $url = printf "%s?%s" $url $query -}}{{- end -}} {{- $text := or (.Get "text") $subject "Add to calendar" -}} {{ $text }}
