{{- /* Shortcode: deeplink-gitlab-mergerequest Renders a link to GitLab's "new merge request" page with prefilled values. Parameters (all named): baseURL — REQUIRED. Project URL, e.g. https://gitlab.com/g/p title — OPTIONAL. Merge request title description — OPTIONAL. Merge request body (Markdown) sourceBranch — OPTIONAL. Branch with the changes (e.g. "feature/login") targetBranch — OPTIONAL. Branch to merge into (e.g. "main") removeSourceBranch — OPTIONAL. "true" to delete the source branch on merge squash — OPTIONAL. "true" to squash commits on merge navSource — OPTIONAL. Free-form navigation source label text — OPTIONAL. Link text class — OPTIONAL. CSS class target — OPTIONAL. Target attribute rel — OPTIONAL. Rel attribute Example: {{< deeplink-gitlab-mergerequest baseURL="https://gitlab.com/g/p" title="Refactor login" sourceBranch="feature/login" targetBranch="main" >}} NOTE: This template mirrors gitlab.MergeRequest() in the Go library and produces the same URL. The `merge_request[...]` keys are written literally, and values use `replace (urlquery .) "+" "%20"` so spaces become %20. If you change one, change the other. */ -}} {{- $baseURL := .Get "baseURL" -}} {{- if not $baseURL -}} {{- errorf "deeplink-gitlab-mergerequest: baseURL is required" -}} {{- end -}} {{- $pairs := slice -}} {{- with .Get "title" -}} {{- $pairs = $pairs | append (printf "merge_request[title]=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- with .Get "description" -}} {{- $pairs = $pairs | append (printf "merge_request[description]=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- with .Get "sourceBranch" -}} {{- $pairs = $pairs | append (printf "merge_request[source_branch]=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- with .Get "targetBranch" -}} {{- $pairs = $pairs | append (printf "merge_request[target_branch]=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- if eq (.Get "removeSourceBranch") "true" -}} {{- $pairs = $pairs | append "merge_request[remove_source_branch]=true" -}} {{- end -}} {{- if eq (.Get "squash") "true" -}} {{- $pairs = $pairs | append "merge_request[squash]=true" -}} {{- end -}} {{- with .Get "navSource" -}} {{- $pairs = $pairs | append (printf "nav_source=%s" (replace (urlquery .) "+" "%20")) -}} {{- end -}} {{- $query := delimit $pairs "&" -}} {{- $url := printf "%s/-/merge_requests/new" (strings.TrimRight "/" $baseURL) -}} {{- if $query -}}{{- $url = printf "%s?%s" $url $query -}}{{- end -}} {{- $text := or (.Get "text") (.Get "title") "Open merge request" -}} {{ $text }}
